cmake_minimum_required(VERSION 3.28)
project(MetalDeviceProbe LANGUAGES CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "This diagnostic requires native macOS")
endif()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Gui)
if(NOT TARGET Qt6::GuiPrivate)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS GuiPrivate)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../lcrenderer/rhi/spike/cmake/LegacyQtMacSdk.cmake")
lc_spike_sdk_has_agl("${Qt6_VERSION}" _agl)
lc_spike_legacy_qt_agl("${Qt6_VERSION}" "${_agl}" _omitted)
add_executable(metal-device-probe probe.mm)
set_target_properties(metal-device-probe PROPERTIES CXX_STANDARD 17 OBJCXX_STANDARD 17)
foreach(framework Metal Foundation IOKit)
    find_library(_${framework} ${framework} REQUIRED)
    target_link_libraries(metal-device-probe PRIVATE "${_${framework}}")
endforeach()
find_library(_objc objc REQUIRED)
target_link_libraries(metal-device-probe PRIVATE Qt6::Gui Qt6::GuiPrivate "${_objc}")
